            estado=Vaga.Estado.ABERTA,

    def para_feed(self):
        """Vagas pÃºblicas, com as das empresas parceiras em destaque."""
        return (
            self.publicas()
            .annotate(
                destaque=Case(
                    When(empresa__tipo=Empresa.Tipo.PARCEIRA, then=Value(1)),
                    default=Value(0),
                    output_field=IntegerField(),
